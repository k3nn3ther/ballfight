import javax.swing.*;
import java.awt.*;

public class UI extends JPanel{

    public UI(){
        initComponents();
    }

    public void initComponents(){
        add(mainPanel());
    }
    public JPanel mainPanel(){
        JPanel panel = new JPanel(null);
        panel.setBackground(new Color(253,246,231));
        panel.add(arena());
        return panel;
    }
    public JPanel arena(){
        JPanel panel = new JPanel(null);
        panel.setBackground(new Color(255,255,255));
        panel.setBounds(20,150,440,440);
        panel.setBorder(BorderFactory.createLineBorder(new Color(0,0,0), 4));
        return panel;
    }
}
